# One Java regex per line. Used by SecretScanTest. Lines starting with # are ignored.
# (?<![a-z]) instead of \b so prefixed names such as TDA_DB_PASSWORD= and TDA_API_CLIENT_SECRET= are caught.
(?i)(?<![a-z])(password|passwd|pwd)\s*[=:]\s*[^\s;\"'<>${}]{4,}
(?i)jdbc:sqlserver://[^\s\"']*;\s*password\s*=
(?i)\b(server|data source)\s*=[^;\n]+;[^\n]*\b(password|pwd)\s*=
\bAKIA[0-9A-Z]{16}\b
-----BEGIN [A-Z ]*PRIVATE KEY-----
(?i)\bbearer\s+[A-Za-z0-9._~+/-]{20,}=*
\beyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]+
(?i)(?<![a-z])(client_secret|clientsecret|api[_-]?key|secret)\s*[=:]\s*[\"']?[^\s\"']{12,}
(?i)\bAccountKey=[A-Za-z0-9+/=]{20,}
